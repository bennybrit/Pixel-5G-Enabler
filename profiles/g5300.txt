# Modem profile for the Exynos 5300 modem: Pixel 9a, and Pixel 7 / 7 Pro / 7a / Fold, 8 / 8 Pro / 8a (same modem;
# untested on those).
# Each [section] is one step, applied in this order at every modem start. Section = SIM slot scope.
# Values: integers (hex or decimal). Data-centric usage comes from Android on this modem (carrier settings of the
# module), so only the 5G SA switch, the NR band mask and the regional band tables are set. Without the switch in the
# profile, Android sets it at runtime and the modem only reads it at its next start. The 9a's stock band mask already
# allows n78; the mask is set for phones whose stock value does not (as on the Pixel 6 and 9).

[sim1]  # 5G SA + NSA and the NR bands (n48/n77/n78 included)
NR.CONFIG.MODE = 0x11
UECAPA_NR_RF_BAND_FR1_BM = 0xc80a

[sim2]  # the same for SIM slot 2
DS.NR.CONFIG.MODE = 0x11
UECAPA_NR_RF_BAND_FR1_BM_DS = 0xc80a

[sim1]  # no regional band tables: with them the SIM's country code limits the bands
OEM_GFEATURE_ENABLE_FCC_BANDS = 0x0

[sim2]  # same for SIM slot 2
OEM_GFEATURE_ENABLE_FCC_BANDS_DS = 0x0
